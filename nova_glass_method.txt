
    def _paint_glass_bar(self, event):
        painter = QPainter(self)
        painter.setRenderHint(QPainter.RenderHint.Antialiasing, True)
        painter.setRenderHint(QPainter.RenderHint.SmoothPixmapTransform, True)

        w = self.width()
        h = self.height()
        t = time.perf_counter() - self.started

        # Read Nova's current state without changing the backend.
        try:
            status = self.status_file.read_text(
                encoding="utf-8", errors="replace"
            ).strip().lower()
        except OSError:
            status = ""

        if any(word in status for word in ("speak", "response", "output")):
            state = "speaking"
        elif any(word in status for word in ("listen", "record", "microphone")):
            state = "listening"
        elif any(word in status for word in ("think", "process", "working")):
            state = "thinking"
        else:
            state = "idle"

        # State-specific movement speeds (rotations per minute).
        rpm = {
            "idle": 17.0,
            "listening": 28.0,
            "thinking": 72.0,
            "speaking": 32.0,
        }[state]
        angle = t * rpm * (2.0 * math.pi / 60.0)

        pulse = (1.0 + math.sin(t * 2.0 * math.pi * 0.30)) * 0.5
        breathe = 0.95 + 0.10 * pulse

        painter.save()
        painter.translate(w / 2.0, h / 2.0)
        painter.scale(breathe, breathe)
        painter.translate(-w / 2.0, -h / 2.0)

        # Soft ambient glow behind the glass capsule.
        glow = QRadialGradient(w * 0.50, h * 0.52, w * 0.53)
        glow.setColorAt(0.0, QColor(0, 229, 255, 25 + int(pulse * 20)))
        glow.setColorAt(0.55, QColor(0, 191, 165, 12))
        glow.setColorAt(1.0, QColor(0, 0, 0, 0))
        painter.setPen(Qt.PenStyle.NoPen)
        painter.setBrush(QBrush(glow))
        painter.drawRoundedRect(0, 0, w, h, 48, 48)

        # Main black stadium-shaped glass body.
        body = QPainterPath()
        body.addRoundedRect(22.0, 43.0, 346.0, 84.0, 42.0, 42.0)

        painter.setBrush(QColor(4, 7, 13, 242))
        painter.setPen(QPen(QColor(0, 229, 255, 90 + int(pulse * 70)), 1.4))
        painter.drawPath(body)

        # Subtle top-edge reflection.
        reflection = QPainterPath()
        reflection.moveTo(54.0, 53.0)
        reflection.cubicTo(105.0, 43.0, 278.0, 43.0, 334.0, 53.0)
        painter.setBrush(Qt.BrushStyle.NoBrush)
        painter.setPen(QPen(QColor(255, 255, 255, 35), 1.0))
        painter.drawPath(reflection)

        # Möbius ribbon and controls will be added in the next chunk.

        # State-specific colours.
        if state == "thinking":
            colours = [
                QColor(213, 0, 249),
                QColor(0, 229, 255),
                QColor(0, 191, 165),
            ]
        elif state == "listening":
            colours = [
                QColor(0, 229, 255),
                QColor(0, 255, 200),
                QColor(0, 160, 255),
            ]
        elif state == "speaking":
            colours = [
                QColor(0, 255, 220),
                QColor(0, 229, 255),
                QColor(213, 0, 249),
            ]
        else:
            colours = [
                QColor(0, 229, 255),
                QColor(0, 191, 165),
                QColor(40, 120, 255),
            ]

        # Render a twisted strip with depth shading.
        # The strip is intentionally confined to the left side of the capsule.
        cx, cy = 116.0, 84.0
        major_radius = 37.0
        minor_radius = 10.0
        segments = 120
        across = 8

        def ribbon_point(u, v):
            a = u + angle
            half = a * 0.5
            radial = major_radius + v * minor_radius * math.cos(half)
            x = cx + radial * math.cos(a)
            y = cy + v * minor_radius * math.sin(half) * math.cos(a)
            z = v * minor_radius * math.sin(half)
            return x, y, z

        # Draw neighbouring strip quads with translucent colour layers.
        for i in range(segments):
            u0 = (2.0 * math.pi * i) / segments
            u1 = (2.0 * math.pi * (i + 1)) / segments

            for j in range(across):
                v0 = -1.0 + 2.0 * j / across
                v1 = -1.0 + 2.0 * (j + 1) / across

                p0 = ribbon_point(u0, v0)
                p1 = ribbon_point(u1, v0)
                p2 = ribbon_point(u1, v1)
                p3 = ribbon_point(u0, v1)

                depth = (p0[2] + p1[2] + p2[2] + p3[2]) / 4.0
                shade = max(0.25, min(1.0, 0.72 + depth / 24.0))

                blend = (i / segments + t * 0.06) % 1.0
                if blend < 0.5:
                    mix = blend * 2.0
                    base = colours[0]
                    other = colours[1]
                else:
                    mix = (blend - 0.5) * 2.0
                    base = colours[1]
                    other = colours[2]

                r = int((base.red() * (1.0 - mix) + other.red() * mix) * shade)
                g = int((base.green() * (1.0 - mix) + other.green() * mix) * shade)
                b = int((base.blue() * (1.0 - mix) + other.blue() * mix) * shade)

                path = QPainterPath()
                path.moveTo(p0[0], p0[1])
                path.lineTo(p1[0], p1[1])
                path.lineTo(p2[0], p2[1])
                path.lineTo(p3[0], p3[1])
                path.closeSubpath()

                alpha = 95 + int(100 * shade)
                painter.setPen(Qt.PenStyle.NoPen)
                painter.setBrush(QColor(r, g, b, alpha))
                painter.drawPath(path)

        # Bright centreline gives the ribbon a glassy neon edge.
        centreline = QPainterPath()
        for i in range(segments + 1):
            u = (2.0 * math.pi * i) / segments
            x, y, _ = ribbon_point(u, 0.0)
            if i == 0:
                centreline.moveTo(x, y)
            else:
                centreline.lineTo(x, y)

        painter.setBrush(Qt.BrushStyle.NoBrush)
        painter.setPen(QPen(QColor(0, 229, 255, 45), 7.0))
        painter.drawPath(centreline)
        painter.setPen(QPen(QColor(120, 255, 245, 210), 1.4))
        painter.drawPath(centreline)

        # NOVA label, kept subtle so the ribbon remains the focus.
        painter.setPen(QColor(175, 210, 225, 175))
        painter.drawText(43, 116, "N O V A")

        # Recessed glass control capsule.
        control = QPainterPath()
        control.addRoundedRect(245.0, 66.0, 94.0, 38.0, 19.0, 19.0)

        painter.setBrush(QColor(12, 19, 29, 248))
        painter.setPen(QPen(QColor(90, 220, 255, 75 + int(pulse * 65)), 1.2))
        painter.drawPath(control)

        # Five-element waveform: four bars plus a central dot.
        # State-based animation until real audio amplitude is connected.
        if state == "listening":
            wave_speed = 13.0
            wave_strength = 1.0
        elif state == "thinking":
            wave_speed = 9.0
            wave_strength = 0.7
        elif state == "speaking":
            wave_speed = 16.0
            wave_strength = 1.0
        else:
            wave_speed = 2.0
            wave_strength = 0.20

        baseline = 85.0
        bar_x = [276.0, 284.0, 292.0, 300.0]

        for index, x in enumerate(bar_x):
            wave = abs(math.sin(t * wave_speed + index * 0.85))
            bar_height = 4.0 + wave * 13.0 * wave_strength
            if state == "idle":
                bar_height = 4.0 + wave * 3.0

            painter.setPen(Qt.PenStyle.NoPen)
            painter.setBrush(
                QColor(0, 229, 255, 130 + int(wave * 100))
            )
            painter.drawRoundedRect(
                int(x), int(baseline - bar_height / 2.0),
                3, max(3, int(bar_height)), 1.5, 1.5
            )

        # Centre dot between waveform and microphone.
        painter.setBrush(QColor(0, 255, 210, 220))
        painter.drawEllipse(QPointF(270.0, 85.0), 2.0, 2.0)

        # Microphone body and stand.
        mic_glow = 100 + int(pulse * 100)
        mic_colour = QColor(0, 229, 255, mic_glow)

        painter.setBrush(Qt.BrushStyle.NoBrush)
        painter.setPen(QPen(mic_colour, 1.8))
        painter.drawRoundedRect(312.0, 74.0, 8.0, 15.0, 4.0, 4.0)

        mic_arc = QPainterPath()
        mic_arc.moveTo(308.0, 83.0)
        mic_arc.lineTo(308.0, 86.0)
        mic_arc.quadTo(308.0, 93.0, 316.0, 93.0)
        mic_arc.quadTo(324.0, 93.0, 324.0, 86.0)
        mic_arc.lineTo(324.0, 83.0)
        painter.drawPath(mic_arc)

        painter.drawLine(316.0, 93.0, 316.0, 97.0)
        painter.drawLine(311.0, 97.0, 321.0, 97.0)

        # Stronger outer capsule edge while Nova is speaking.
        if state == "speaking":
            painter.setBrush(Qt.BrushStyle.NoBrush)
            painter.setPen(
                QPen(QColor(0, 229, 255, 85 + int(pulse * 110)), 1.6)
            )
            painter.drawRoundedRect(22.0, 43.0, 346.0, 84.0, 42.0, 42.0)

        painter.restore()
        painter.end()
